<script>

document.addEventListener('DOMContentLoaded', function() {


  const calendar = new FullCalendar.Calendar(document.getElementById('calendar'), {

    initialView: 'dayGridMonth',

    locale: 'ja',


    eventSources: [

      {

        url: 'https://example.com/your_calendar.ics', // ←ここ差し替え

        format: 'ics'

      }

    ],


    displayEventTime: false,

    dayMaxEventRows: false,

    expandRows: true,

    height: 'auto',


    eventOrder: 'title',


    /* 色分け（自由に変更） */

    eventDidMount: function(info) {

      const t = info.event.title;


      if (t.includes('LIVE')) info.el.style.backgroundColor = '#ff6b6b';

      else if (t.includes('イベント')) info.el.style.backgroundColor = '#4dabf7';

      else if (t.includes('休')) info.el.style.backgroundColor = '#ffd43b';

      else info.el.style.backgroundColor = '#adb5bd';

    }


  });


  calendar.render();

});

</script>
